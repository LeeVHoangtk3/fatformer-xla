## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Cài đặt `training/checkpoint_manager.py`:
```python
import os
import shutil
import torch

class CheckpointManager:
    def __init__(self, drive_dir="/content/drive/MyDrive/FatFormer_Hub/checkpoints", 
                 local_dir="/content/checkpoints", keep_best=3):
        self.drive_dir = drive_dir
        self.local_dir = local_dir
        self.keep_best = keep_best
        self.best_metrics = [] # Danh sách lưu [(score, filepath)]
        
        os.makedirs(drive_dir, exist_ok=True)
        os.makedirs(local_dir, exist_ok=True)

    def save_checkpoint(self, state_dict, epoch, metric_score, is_best=False):
        """Lưu checkpoint và đồng bộ ngay sang Google Drive 5TB."""
        filename = f"checkpoint_epoch_{epoch:02d}_val_{metric_score:.2f}.pth"
        local_path = os.path.join(self.local_dir, filename)
        drive_path = os.path.join(self.drive_dir, filename)
        
        # 1. Lưu tại local SSD Colab
        torch.save({
            "epoch": epoch,
            "state_dict": state_dict,
            "metric_score": metric_score
        }, local_path)
        
        # 2. Sao lưu tức thời sang Drive 5TB
        shutil.copy(local_path, drive_path)
        
        # 3. Luôn cập nhật file last.pth để phục vụ resume
        last_local = os.path.join(self.local_dir, "last.pth")
        last_drive = os.path.join(self.drive_dir, "last.pth")
        shutil.copy(local_path, last_local)
        shutil.copy(local_path, last_drive)
        
        print(f"[✓] Đã sao lưu Checkpoint Epoch {epoch} an toàn sang Drive 5TB: {drive_path}")

    def load_latest_for_resume(self, model, optimizer=None):
        """Khôi phục trạng thái mô hình từ last.pth trên Drive nếu bị ngắt kết nối."""
        last_drive = os.path.join(self.drive_dir, "last.pth")
        if not os.path.exists(last_drive):
            print("[*] Không tìm thấy checkpoint cũ, bắt đầu huấn luyện từ đầu (Epoch 1).")
            return 1, model, optimizer
            
        print(f"[*] Đang khôi phục phiên làm việc từ: {last_drive}...")
        ckpt = torch.load(last_drive, map_location="cpu")
        model.load_state_dict(ckpt["state_dict"])
        resume_epoch = ckpt["epoch"] + 1
        print(f"[✓] Khôi phục thành công! Sẽ tiếp tục huấn luyện từ Epoch {resume_epoch}.")
        return resume_epoch, model, optimizer
```

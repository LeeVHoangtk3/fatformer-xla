### Bước 1: Module [`evaluation/metrics.py`](file:///d:/GIT%20REPO/.nam4/fatformer-xla/evaluation/metrics.py)
```python
import numpy as np
from sklearn.metrics import accuracy_score, average_precision_score, roc_auc_score, roc_curve

def calculate_metrics(y_true, y_pred_probs):
    """
    y_true: mảng numpy [N] (0: Real, 1: Fake)
    y_pred_probs: mảng numpy [N] (xác suất dự đoán nhãn 1 - Fake)
    """
    y_true = np.array(y_true)
    y_pred_probs = np.array(y_pred_probs)
    y_pred_labels = (y_pred_probs >= 0.5).astype(int)
    
    # 1. Accuracy
    acc = accuracy_score(y_true, y_pred_labels) * 100.0
    
    # 2. Average Precision (AP)
    ap = average_precision_score(y_true, y_pred_probs) * 100.0
    
    # 3. AUC-ROC
    auc = roc_auc_score(y_true, y_pred_probs) * 100.0
    
    # 4. Equal Error Rate (EER)
    fpr, tpr, thresholds = roc_curve(y_true, y_pred_probs, pos_label=1)
    fnr = 1 - tpr
    eer_idx = np.nanargmin(np.absolute(fnr - fpr))
    eer = (fpr[eer_idx] + fnr[eer_idx]) / 2.0 * 100.0
    
    return {
        "ACC": round(acc, 2),
        "AP": round(ap, 2),
        "AUC": round(auc, 2),
        "EER": round(eer, 2)
    }
```

### Bước 2: Script `evaluation/fast_eval.py`
```python
import os
import torch
import random
from torch.utils.data import DataLoader, Subset
from torchvision.datasets import ImageFolder
from datasets.transforms import get_eval_transforms
from evaluation.metrics import calculate_metrics

def evaluate_subset(model, dataset_path, samples=500, batch_size=32, device="cuda"):
    transforms = get_eval_transforms()
    full_ds = ImageFolder(dataset_path, transform=transforms)
    
    # Lấy mẫu ngẫu nhiên samples ảnh
    indices = random.sample(range(len(full_ds)), min(samples, len(full_ds)))
    sub_ds = Subset(full_ds, indices)
    loader = DataLoader(sub_ds, batch_size=batch_size, shuffle=False, num_workers=2)
    
    y_true, y_probs = [], []
    model.eval()
    with torch.no_grad():
        for imgs, labels in loader:
            imgs = imgs.to(device)
            logits = model(imgs)
            probs = torch.softmax(logits, dim=-1)[:, 1] # Xác suất Fake
            y_true.extend(labels.numpy().tolist())
            y_probs.extend(probs.cpu().numpy().tolist())
            
    return calculate_metrics(y_true, y_probs)
```

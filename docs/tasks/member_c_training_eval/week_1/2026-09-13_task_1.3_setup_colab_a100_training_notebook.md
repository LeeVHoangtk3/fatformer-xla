## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Cấu trúc mã nguồn trong `notebooks/train.ipynb`:
```python
import os
import torch
import torch.nn as nn
from torch.cuda.amp import autocast, GradScaler
from models.fatformer import build_fatformer
from datasets.transforms import get_train_transforms
from torch.utils.data import DataLoader
from torchvision.datasets import ImageFolder

# 1. Kiểm tra phần cứng GPU
assert torch.cuda.is_available(), "Vui lòng bật GPU trong Runtime -> Change runtime type!"
gpu_name = torch.cuda.get_device_name(0)
print(f"[✓] Đang chạy trên GPU: {gpu_name} (VRAM: {torch.cuda.get_device_properties(0).total_memory / 1e9:.2f} GB)")

# 2. Khởi tạo mô hình & AMP Scaler
device = torch.device("cuda")
model = build_fatformer(pretrained_clip="ViT-L-14.pt").to(device)
scaler = GradScaler() # Hỗ trợ AMP FP16 chống underflow

# 3. Cấu hình Optimizer & Loss
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.AdamW(filter(lambda p: p.requires_grad, model.parameters()), 
                             lr=2e-4, weight_decay=1e-4)

# 4. DataLoader với Dummy Dataset
dataset = ImageFolder("/content/dataset_local", transform=get_train_transforms())
loader = DataLoader(dataset, batch_size=32, shuffle=True, num_workers=2, pin_memory=True)

# 5. Chạy thử 1 Step Kiểm Tra VRAM & Gradient
model.train()
optimizer.zero_grad()

for step, (images, labels) in enumerate(loader):
    images, labels = images.to(device, non_blocking=True), labels.to(device, non_blocking=True)
    
    # Bật Autocast FP16
    with autocast():
        outputs = model(images)
        loss = criterion(outputs, labels) / 2 # Chia cho gradient_accumulation_steps=2
        
    scaler.scale(loss).backward()
    
    if (step + 1) % 2 == 0:
        scaler.step(optimizer)
        scaler.update()
        optimizer.zero_grad()
        
    allocated_vram = torch.cuda.max_memory_allocated() / 1e9
    print(f"[✓] Step {step+1} thành công! Loss: {loss.item()*2:.4f} | Đỉnh VRAM tiêu thụ: {allocated_vram:.2f} GB / 40.0 GB")
    break # Dừng ngay sau 1 step để hoàn tất sanity check
```

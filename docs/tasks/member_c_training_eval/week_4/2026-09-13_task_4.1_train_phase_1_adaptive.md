## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Cấu hình Vòng lặp Huấn luyện trong `train.ipynb`:
```python
import time
import torch
from torch.cuda.amp import autocast, GradScaler
from training.checkpoint_manager import CheckpointManager
from evaluation.fast_eval import evaluate_subset

# Thiết lập tham số Pha 1
EPOCHS = 4
LR_ADAPTER = 2e-4
LR_GATING = 1e-3
ACCUM_STEPS = 2 # Effective batch size = 32 * 2 = 64

# Optimizer phân tầng tốc độ học (Discriminative LR)
param_groups = [
    {"params": [p for n, p in model.named_parameters() if "freq_gating" in n and p.requires_grad], "lr": LR_GATING},
    {"params": [p for n, p in model.named_parameters() if "freq_gating" not in n and p.requires_grad], "lr": LR_ADAPTER}
]
optimizer = torch.optim.AdamW(param_groups, weight_decay=1e-4)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=EPOCHS*len(train_loader)//ACCUM_STEPS)
scaler = GradScaler()
ckpt_mgr = CheckpointManager()

print(f"[*] Bắt đầu Huấn luyện Pha 1 (Adaptive Generalization) - {EPOCHS} Epochs...")

for epoch in range(1, EPOCHS + 1):
    model.train()
    total_loss = 0.0
    start_epoch = time.time()
    optimizer.zero_grad()
    
    for step, (images, labels) in enumerate(train_loader):
        images = images.to(device, non_blocking=True)
        labels = labels.to(device, non_blocking=True)
        
        with autocast():
            logits = model(images)
            loss = criterion(logits, labels) / ACCUM_STEPS
            
        scaler.scale(loss).backward()
        total_loss += loss.item() * ACCUM_STEPS
        
        if (step + 1) % ACCUM_STEPS == 0:
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad()
            scheduler.step()
            
        if (step + 1) % 100 == 0:
            print(f"Epoch [{epoch}/{EPOCHS}] | Step [{step+1}/{len(train_loader)}] | Loss: {loss.item()*ACCUM_STEPS:.4f}")
            
    epoch_duration = time.time() - start_epoch
    avg_loss = total_loss / len(train_loader)
    print(f"[✓] Hoàn tất Epoch {epoch} sau {epoch_duration/60:.2f} phút | Avg Loss: {avg_loss:.4f}")
    
    # Đánh giá nhanh Validation bằng Fast-Eval
    val_res = evaluate_subset(model, "/content/dataset_local/test_clean/progan", samples=500, device=device)
    print(f"[*] Validation ACC: {val_res['ACC']}% | AP: {val_res['AP']}%")
    
    # Tự động sao lưu checkpoint về Drive 5TB
    ckpt_mgr.save_checkpoint(model.state_dict(), epoch, val_res['ACC'])
```

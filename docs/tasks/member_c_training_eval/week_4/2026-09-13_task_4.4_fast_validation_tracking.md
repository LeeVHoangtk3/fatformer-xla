### Cài đặt Hook Đánh giá trong Vòng lặp:
```python
import json
import matplotlib.pyplot as plt

history = {
    "train_loss": [],
    "val_clean_acc": [],
    "val_degraded_acc": []
}

def on_epoch_end(epoch, train_loss, model, device="cuda"):
    print(f"\n[*] Đang chạy Fast-Validation cho Epoch {epoch}...")
    
    # 1. Đánh giá Clean
    res_clean = evaluate_subset(model, "/content/dataset_local/val_clean", samples=500, device=device)
    # 2. Đánh giá Degraded Q=50
    res_deg = evaluate_subset(model, "/content/dataset_local/val_degraded_q50", samples=500, device=device)
    
    history["train_loss"].append(train_loss)
    history["val_clean_acc"].append(res_clean["ACC"])
    history["val_degraded_acc"].append(res_deg["ACC"])
    
    print(f"[Epoch {epoch}] Loss: {train_loss:.4f} | Clean ACC: {res_clean['ACC']}% | Degraded ACC: {res_deg['ACC']}%")
    
    # Lưu file json
    with open("logs/history.json", "w") as f:
        json.dump(history, f, indent=4)
        
    # Cập nhật hình ảnh đồ thị hội tụ
    plt.figure(figsize=(10, 4))
    plt.subplot(1, 2, 1)
    plt.plot(history["train_loss"], label="Train Loss", color="red")
    plt.title("Hội Tụ Hàm Mất Mát (Loss Curve)")
    plt.xlabel("Epoch")
    plt.grid(True)
    
    plt.subplot(1, 2, 2)
    plt.plot(history["val_clean_acc"], label="Clean ACC", color="blue")
    plt.plot(history["val_degraded_acc"], label="Degraded ACC", color="green")
    plt.title("Độ Chính Xác Validation (ACC)")
    plt.xlabel("Epoch")
    plt.legend()
    plt.grid(True)
    
    plt.tight_layout()
    plt.savefig("docs/figures/training_curves.png", dpi=300)
    plt.close()
```

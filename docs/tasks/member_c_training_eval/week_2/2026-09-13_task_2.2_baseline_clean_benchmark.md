## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Viết script `evaluation/benchmark_clean.py`:
```python
import os
import json
import torch
import pandas as pd
from models.fatformer import build_fatformer
from evaluation.fast_eval import evaluate_subset

def run_clean_benchmark():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"[*] Khởi tạo FatFormer gốc trên {device}...")
    model = build_fatformer(pretrained_clip="ViT-L-14.pt").to(device)
    ckpt = torch.load("fatformer_4class_ckpt.pth", map_location=device)
    model.load_state_dict(ckpt["model"], strict=True)
    model.eval()

    test_root = "/content/dataset_local/test_clean"
    subsets = [d for d in os.listdir(test_root) if os.path.isdir(os.path.join(test_root, d))]
    
    results = {}
    table_rows = []
    
    for sub in subsets:
        sub_path = os.path.join(test_root, sub)
        # Sử dụng 1000 mẫu đại diện hoặc toàn bộ tập test
        res = evaluate_subset(model, sub_path, samples=1000, device=device)
        results[sub] = res
        table_rows.append({"Generator": sub, **res})
        print(f"[✓] {sub}: ACC={res['ACC']}% | AP={res['AP']}% | AUC={res['AUC']}%")
        
    df = pd.DataFrame(table_rows)
    print("\n--- BẢNG KẾT QUẢ BASELINE CLEAN TỔNG HỢP ---")
    print(df.to_markdown(index=False))
    
    # Lưu kết quả
    with open("docs/paper_analysis/baseline_clean_results.json", "w") as f:
        json.dump(results, f, indent=4)
        
    with open("docs/paper_analysis/baseline_clean_results.md", "w") as f:
        f.write("# KẾT QUẢ TÁI LẬP BASELINE GỐC TRÊN TẬP CLEAN\n\n")
        f.write(df.to_markdown(index=False))

if __name__ == "__main__":
    run_clean_benchmark()
```

## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Viết script `evaluation/benchmark_degraded.py`:
```python
import os
import json
import torch
import pandas as pd
from models.fatformer import build_fatformer
from evaluation.fast_eval import evaluate_subset

def run_degraded_benchmark():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    model = build_fatformer(pretrained_clip="ViT-L-14.pt").to(device)
    ckpt = torch.load("fatformer_4class_ckpt.pth", map_location=device)
    model.load_state_dict(ckpt["model"], strict=True)
    model.eval()

    degraded_root = "/content/dataset_local/test_degraded"
    conditions = ["jpeg_q70", "jpeg_q50", "jpeg_q30", "blur_s2", "down_up"]
    
    summary_rows = []
    for cond in conditions:
        cond_dir = os.path.join(degraded_root, cond)
        if not os.path.exists(cond_dir):
            continue
            
        print(f"[*] Đang đánh giá trạng thái suy biến: {cond}...")
        # Đánh giá trên tập con tổng hợp hoặc quét từng generator
        res = evaluate_subset(model, cond_dir, samples=1500, device=device)
        summary_rows.append({"Condition": cond, **res})
        print(f"[!] {cond}: ACC={res['ACC']}% | AP={res['AP']}% (Sụt giảm nặng)")

    df = pd.DataFrame(summary_rows)
    print("\n--- BẢNG ĐỐI CHỨNG BASELINE SUY BIẾN ---")
    print(df.to_markdown(index=False))
    
    with open("docs/paper_analysis/baseline_degraded_results.md", "w") as f:
        f.write("# BẢNG SỐ LIỆU SỤT GIẢM HIỆU NĂNG CỦA FATFORMER GỐC TRƯỚC SUY BIẾN\n\n")
        f.write(df.to_markdown(index=False))

if __name__ == "__main__":
    run_degraded_benchmark()
```

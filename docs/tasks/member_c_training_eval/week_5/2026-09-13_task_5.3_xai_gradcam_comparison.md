## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Viết script tạo ảnh so sánh ghép đôi:
Tạo `tools/generate_gradcam_comparison.py`:
```python
import os
import cv2
import torch
import matplotlib.pyplot as plt
from tools.visualize_cam import FatFormerGradCAM, overlay_cam
from models.fatformer import build_fatformer, build_fatformer_srm
from datasets.transforms import get_eval_transforms
from PIL import Image

def generate_comparison_grid(sample_paths, out_path="docs/figures/gradcam_comparison_grid.png"):
    device = "cuda" if torch.cuda.is_available() else "cpu"
    
    # 1. Nạp mô hình gốc
    model_base = build_fatformer().to(device)
    ckpt_base = torch.load("fatformer_4class_ckpt.pth", map_location=device)
    model_base.load_state_dict(ckpt_base["model"])
    cam_base = FatFormerGradCAM(model_base, model_base.visual.transformer.resblocks[-1])
    
    # 2. Nạp mô hình đề xuất
    model_prop = build_fatformer_srm().to(device)
    ckpt_prop = torch.load("checkpoints/fatformer_srm_robust_final.pth", map_location=device)
    model_prop.load_state_dict(ckpt_prop["model"] if "model" in ckpt_prop else ckpt_prop["state_dict"])
    cam_prop = FatFormerGradCAM(model_prop, model_prop.visual.transformer.resblocks[-1])
    
    transforms = get_eval_transforms()
    n_samples = len(sample_paths)
    fig, axes = plt.subplots(n_samples, 3, figsize=(12, 4 * n_samples))
    
    col_titles = ["Ảnh Đầu Vào (JPEG Q=30)", "FatFormer Gốc (Bị Nhiễu Nén)", "Mô Hình Đề Xuất (Tập Trung Đúng)"]
    for j, title in enumerate(col_titles):
        axes[0, j].set_title(title, fontsize=14, weight='bold', pad=10)
        
    for i, p in enumerate(sample_paths):
        raw_img = Image.open(p).convert("RGB")
        tensor = transforms(raw_img).unsqueeze(0).to(device)
        
        # Tạo heatmap
        hm_base = cam_base.generate_heatmap(tensor)
        hm_prop = cam_prop.generate_heatmap(tensor)
        
        # Hiển thị
        axes[i, 0].imshow(raw_img)
        axes[i, 0].axis("off")
        
        axes[i, 1].imshow(raw_img)
        axes[i, 1].imshow(cv2.resize(hm_base, raw_img.size), cmap="jet", alpha=0.5)
        axes[i, 1].axis("off")
        
        axes[i, 2].imshow(raw_img)
        axes[i, 2].imshow(cv2.resize(hm_prop, raw_img.size), cmap="jet", alpha=0.5)
        axes[i, 2].axis("off")
        
    plt.tight_layout()
    plt.savefig(out_path, dpi=300, bbox_inches="tight")
    print(f"[✓] Đã xuất lưới ảnh so sánh Grad-CAM tại: {out_path}")

if __name__ == "__main__":
    test_samples = [
        "dataset_local/test_degraded/jpeg_q30/progan/1_fake/sample_01.jpg",
        "dataset_local/test_degraded/jpeg_q30/midjourney/1_fake/sample_02.jpg",
        "dataset_local/test_degraded/jpeg_q30/sd_v15/1_fake/sample_03.jpg",
        "dataset_local/test_degraded/jpeg_q30/progan/0_real/sample_04.jpg"
    ]
    generate_comparison_grid(test_samples)
```

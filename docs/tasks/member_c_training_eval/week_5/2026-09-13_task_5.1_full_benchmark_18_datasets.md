## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Viết script `evaluation/full_benchmark_matrix.py`:
```python
import os
import json
import torch
import pandas as pd
from models.fatformer import build_fatformer_srm
from evaluation.fast_eval import evaluate_subset

def run_full_benchmark():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"[*] Bắt đầu Full-Benchmark trên GPU {device}...")
    
    # 1. Nạp mô hình hoàn chỉnh
    model = build_fatformer_srm(pretrained_clip="ViT-L-14.pt").to(device)
    ckpt = torch.load("checkpoints/fatformer_srm_robust_final.pth", map_location=device)
    model.load_state_dict(ckpt["model"] if "model" in ckpt else ckpt["state_dict"])
    model.eval()
    
    test_clean_root = "/content/dataset_local/test_clean"
    generators = sorted([d for d in os.listdir(test_clean_root) if os.path.isdir(os.path.join(test_clean_root, d))])
    
    master_records = []
    
    for gen in generators:
        print(f"\n================ Đánh giá Generator: {gen} ================")
        clean_path = os.path.join(test_clean_root, gen)
        q70_path = f"/content/dataset_local/test_degraded/jpeg_q70/{gen}"
        q50_path = f"/content/dataset_local/test_degraded/jpeg_q50/{gen}"
        q30_path = f"/content/dataset_local/test_degraded/jpeg_q30/{gen}"
        blur_path = f"/content/dataset_local/test_degraded/blur_s2/{gen}"
        
        # Đánh giá từng trạng thái
        m_clean = evaluate_subset(model, clean_path, samples=1000, device=device)
        m_q70 = evaluate_subset(model, q70_path, samples=1000, device=device) if os.path.exists(q70_path) else {}
        m_q50 = evaluate_subset(model, q50_path, samples=1000, device=device) if os.path.exists(q50_path) else {}
        m_q30 = evaluate_subset(model, q30_path, samples=1000, device=device) if os.path.exists(q30_path) else {}
        m_blur = evaluate_subset(model, blur_path, samples=1000, device=device) if os.path.exists(blur_path) else {}
        
        row = {
            "Generator": gen,
            "Clean ACC": m_clean.get("ACC", "-"),
            "Clean AP": m_clean.get("AP", "-"),
            "Q=70 ACC": m_q70.get("ACC", "-"),
            "Q=50 ACC": m_q50.get("ACC", "-"),
            "Q=30 ACC": m_q30.get("ACC", "-"),
            "Blur ACC": m_blur.get("ACC", "-")
        }
        master_records.append(row)
        print(row)
        
    df = pd.DataFrame(master_records)
    
    # Tính dòng trung bình (Average)
    numeric_cols = [c for c in df.columns if c != "Generator"]
    avg_row = {"Generator": "**AVERAGE**"}
    for c in numeric_cols:
        vals = [float(v) for v in df[c] if v != "-"]
        avg_row[c] = round(sum(vals)/len(vals), 2) if vals else "-"
    df = pd.concat([df, pd.DataFrame([avg_row])], ignore_index=True)
    
    print("\n\n################ BẢNG MA TRẬN FULL BENCHMARK 2 CHIỀU ################")
    print(df.to_markdown(index=False))
    
    with open("docs/paper_analysis/final_full_benchmark_matrix.md", "w") as f:
        f.write("# BẢNG SỐ LIỆU ĐÁNH GIÁ BENCHMARK TOÀN DIỆN CUỐI KỲ\n\n")
        f.write(df.to_markdown(index=False))

if __name__ == "__main__":
    run_full_benchmark()
```

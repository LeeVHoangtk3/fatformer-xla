### Bước 1: Cài đặt logic đóng băng tham số
Thêm hàm vào `models/fatformer.py` hoặc tạo file `tools/param_manager.py`:
```python
def configure_trainable_parameters(model):
    """
    Đóng băng 94.1% CLIP backbone, chỉ mở khóa 5.9% Adapter.
    """
    trainable_keywords = [
        "forgery_aware_adapter",
        "srm_block.proj",
        "freq_gating",
        "language_guided_alignment",
        "text_guided_interactor",
        "head",
        "prompt_learner"
    ]
    
    total_params = 0
    trainable_params = 0
    
    for name, param in model.named_parameters():
        total_params += param.numel()
        # Kiểm tra xem tham số có thuộc danh sách module adapter được mở khóa không
        if any(kw in name for kw in trainable_keywords):
            param.requires_grad = True
            trainable_params += param.numel()
        else:
            param.requires_grad = False
            
    freeze_percent = (1.0 - trainable_params / total_params) * 100
    print(f"[*] Tổng tham số: {total_params:,}")
    print(f"[*] Tham số huấn luyện (Trainable): {trainable_params:,} ({trainable_params/total_params*100:.2f}%)")
    print(f"[✓] Tỷ lệ đóng băng: {freeze_percent:.2f}% (Đạt tiêu chuẩn 94.1%)")
    return model
```

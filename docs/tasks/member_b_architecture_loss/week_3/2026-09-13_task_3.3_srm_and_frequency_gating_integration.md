### Bước 1: Module [`models/srm.py`](file:///d:/GIT%20REPO/.nam4/fatformer-xla/models/srm.py)
```python
import torch
import torch.nn as nn

class SpatialResidualBlock(nn.Module):
    """
    Bộ lọc vết dư không gian SRM gồm 3 kernels pháp y kinh điển.
    Trọng số cố định (requires_grad = False), 0 tốn tham số huấn luyện.
    """
    def __init__(self, in_channels=3, out_dim=768):
        super().__init__()
        # 1. Kernel bậc 1 (dò vi sai liền kề)
        k1 = torch.tensor([
            [0., 0., 0.],
            [-1., 1., 0.],
            [0., 0., 0.]
        ])
        
        # 2. Kernel bậc 2 (Laplacian 4 hướng)
        k2 = torch.tensor([
            [0., -1., 0.],
            [-1., 4., -1.],
            [0., -1., 0.]
        ])
        
        # 3. Kernel Square 5x5 (dò vết nội suy bilinear/bicubic)
        k3 = torch.tensor([
            [-1.,  2.,  -2.,  2., -1.],
            [ 2., -6.,   8., -6.,  2.],
            [-2.,  8., -12.,  8., -2.],
            [ 2., -6.,   8., -6.,  2.],
            [-1.,  2.,  -2.,  2., -1.]
        ]) / 12.0
        
        # Đăng ký buffers để PyTorch quản lý thiết bị GPU/CPU
        self.register_buffer("k1", k1.view(1, 1, 3, 3).repeat(in_channels, 1, 1, 1))
        self.register_buffer("k2", k2.view(1, 1, 3, 3).repeat(in_channels, 1, 1, 1))
        self.register_buffer("k3", k3.view(1, 1, 5, 5).repeat(in_channels, 1, 1, 1))
        
        # Lớp chiếu Linear 1x1 ánh xạ về không gian đặc trưng của FAA
        # 3 kernels x 3 channels = 9 channels
        self.proj = nn.Sequential(
            nn.Conv2d(9, out_dim, kernel_size=1, bias=False),
            nn.BatchNorm2d(out_dim),
            nn.GELU()
        )

    def forward(self, x):
        # x: [B, 3, 224, 224]
        # Depthwise convolution cho từng kênh RGB
        res1 = nn.functional.conv2d(x, self.k1, padding=1, groups=3)
        res2 = nn.functional.conv2d(x, self.k2, padding=1, groups=3)
        res3 = nn.functional.conv2d(x, self.k3, padding=2, groups=3)
        
        # Ghép 3 vết dư: [B, 9, 224, 224]
        res_cat = torch.cat([res1, res2, res3], dim=1)
        # Chiếu về kích thước đặc trưng của mô hình: [B, out_dim, 224, 224]
        return self.proj(res_cat)
```

### Bước 2: Module [`models/gating.py`](file:///d:/GIT%20REPO/.nam4/fatformer-xla/models/gating.py)
```python
import torch
import torch.nn as nn

class DynamicFrequencyGating(nn.Module):
    """
    Mạng Gating dự đoán hệ số thích ứng lambda(x) dựa trên mức suy biến ảnh.
    lambda(x) in [0.0, 2.0]
    """
    def __init__(self, in_features=768, hidden_dim=64):
        super().__init__()
        self.mlp = nn.Sequential(
            nn.Linear(in_features, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, 1),
            nn.Sigmoid()
        )

    def forward(self, feature_map):
        # feature_map: [B, C, H, W] -> Global Average Pooling -> [B, C]
        gap = torch.mean(feature_map, dim=[2, 3])
        # Xuất lambda trong khoảng [0, 2.0]
        lambda_val = self.mlp(gap) * 2.0
        return lambda_val # Shape: [B, 1, 1, 1]
```

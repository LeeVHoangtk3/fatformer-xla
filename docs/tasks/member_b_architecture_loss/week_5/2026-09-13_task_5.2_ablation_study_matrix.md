### Viết script tự động trích xuất bảng Markdown:
Tạo `tools/generate_ablation_table.py`:
```python
import json
import pandas as pd

def generate_table():
    models = {
        "V1_Baseline": "logs/eval_v1_baseline.json",
        "V2_AugOnly": "logs/eval_v2_aug.json",
        "V3_SRM": "logs/eval_v3_srm.json",
        "V4_Proposed": "logs/eval_v4_proposed.json"
    }
    
    rows = []
    for m_name, log_path in models.items():
        with open(log_path) as f:
            data = json.load(f)
            rows.append({
                "Model": m_name,
                "Clean": data.get("clean_acc", 0.0),
                "Q70": data.get("q70_acc", 0.0),
                "Q50": data.get("q50_acc", 0.0),
                "Q30": data.get("q30_acc", 0.0),
                "Blur": data.get("blur_acc", 0.0),
            })
            
    df = pd.DataFrame(rows)
    print(df.to_markdown(index=False))
    with open("docs/paper_analysis/ablation_study_results.md", "w") as f:
        f.write("# KẾT QUẢ PHÂN TÍCH TRIỆT TIÊU (ABLATION STUDY)\n\n")
        f.write(df.to_markdown(index=False))

if __name__ == "__main__":
    generate_table()
```

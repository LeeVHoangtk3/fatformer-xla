## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Cài đặt script `tools/analyze_gating.py`:
```python
import torch
import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from tqdm import tqdm
from models.fatformer import build_fatformer_srm
from datasets.transforms import get_eval_transforms
from torch.utils.data import DataLoader
from torchvision.datasets import ImageFolder

def evaluate_lambda_distribution(model, data_dir, device="cuda"):
    transforms = get_eval_transforms()
    dataset = ImageFolder(data_dir, transform=transforms)
    loader = DataLoader(dataset, batch_size=32, shuffle=False, num_workers=2)
    
    lambdas = []
    with torch.no_grad():
        for imgs, _ in tqdm(loader, desc=f"Evaluating {data_dir}"):
            imgs = imgs.to(device)
            # Hàm forward có cờ return_lambda=True để lấy giá trị gating
            _, lambda_val = model(imgs, return_lambda=True)
            lambdas.extend(lambda_val.squeeze().cpu().numpy().tolist())
    return lambdas

def main():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    model = build_fatformer_srm().to(device)
    ckpt = torch.load("checkpoints/fatformer_srm_robust_final.pth", map_location=device)
    model.load_state_dict(ckpt["model"])
    model.eval()
    
    test_conditions = {
        "Clean": "dataset_local/test_clean",
        "JPEG Q=70": "dataset_local/test_degraded/jpeg_q70",
        "JPEG Q=50": "dataset_local/test_degraded/jpeg_q50",
        "JPEG Q=30": "dataset_local/test_degraded/jpeg_q30",
        "Blur s=2.0": "dataset_local/test_degraded/blur_s2"
    }
    
    records = []
    for cond_name, path in test_conditions.items():
        vals = evaluate_lambda_distribution(model, path, device)
        for v in vals:
            records.append({"Condition": cond_name, "Lambda": v})
            
    df = pd.DataFrame(records)
    
    # Vẽ biểu đồ Boxplot
    plt.figure(figsize=(9, 5))
    sns.boxplot(x="Condition", y="Lambda", data=df, palette="Blues_r")
    plt.title("Phân Bố Giá Trị Hệ Số Thích Ứng Cổng Tần Số $\lambda(x)$ Theo Mức Suy Biến", fontsize=13)
    plt.ylabel("Giá trị $\lambda(x)$ (Frequency Weight)")
    plt.grid(axis='y', linestyle='--', alpha=0.7)
    plt.savefig("docs/figures/lambda_distribution_analysis.png", dpi=300, bbox_inches="tight")
    print("[✓] Đã xuất biểu đồ phân tích gating tại docs/figures/lambda_distribution_analysis.png!")

if __name__ == "__main__":
    main()
```

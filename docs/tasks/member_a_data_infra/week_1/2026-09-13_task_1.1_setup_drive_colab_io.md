### Bước 3: Viết Notebook `notebooks/00_setup_env.ipynb`
Tạo notebook và cài đặt hàm giải nén tiêu chuẩn:
```python
import os
import shutil
import tarfile
import time
from google.colab import drive

# 1. Mount Drive (Chỉ tốn 0 byte trên tài khoản Colab Pro)
drive.mount('/content/drive')

DRIVE_HUB = "/content/drive/MyDrive/FatFormer_Hub"
LOCAL_DIR = "/content/dataset_local"

def extract_to_local(tar_filename, target_dir=LOCAL_DIR):
    """
    Sao chép tệp .tar từ Google Drive sang SSD Colab và giải nén cục bộ.
    Tránh hiện tượng nghẽn cổ chai I/O khi DataLoader đọc trực tiếp từ Drive.
    """
    start_time = time.time()
    tar_path = os.path.join(DRIVE_HUB, "datasets", tar_filename)
    local_tar = os.path.join("/content", tar_filename)
    
    print(f"[*] Đang sao chép {tar_filename} sang SSD Colab...")
    shutil.copy(tar_path, local_tar)
    
    print(f"[*] Đang giải nén vào {target_dir}...")
    os.makedirs(target_dir, exist_ok=True)
    with tarfile.open(local_tar, "r") as tar:
        tar.extractall(target_dir)
        
    os.remove(local_tar) # Dọn dẹp tệp nén tạm để giải phóng SSD
    print(f"[✓] Hoàn tất sau {time.time() - start_time:.2f} giây! Dữ liệu sẵn sàng tại SSD nội bộ.")

# Chạy thử nghiệm với Dummy Dataset
extract_to_local("dummy_data.tar")
```

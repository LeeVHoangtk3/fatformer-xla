## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Bước 1: Viết script `data/make_degraded.py`
Tạo file [`data/make_degraded.py`](file:///d:/GIT%20REPO/.nam4/fatformer-xla/data/make_degraded.py) với logic xử lý ảnh:
```python
import os
import io
import argparse
from PIL import Image, ImageFilter
from tqdm import tqdm

def apply_jpeg_compression(image: Image.Image, quality: int) -> Image.Image:
    """Nén JPEG theo bảng lượng tử hóa tương ứng hệ số quality."""
    buffer = io.BytesIO()
    image.save(buffer, format="JPEG", quality=quality)
    buffer.seek(0)
    return Image.open(buffer).copy()

def apply_gaussian_blur(image: Image.Image, sigma: float) -> Image.Image:
    """Làm mờ ảnh bằng bộ lọc Gaussian với bán kính sigma."""
    return image.filter(ImageFilter.GaussianBlur(radius=sigma))

def apply_down_up_sample(image: Image.Image, intermediate_size=(112, 112)) -> Image.Image:
    """Mô phỏng nén resize mạng xã hội: thu nhỏ rồi phóng to lại."""
    orig_size = image.size
    down = image.resize(intermediate_size, Image.Resampling.BICUBIC)
    up = down.resize(orig_size, Image.Resampling.BICUBIC)
    return up

def process_dataset(input_dir, output_root):
    """Quét toàn bộ ảnh trong input_dir và sinh các phiên bản degraded."""
    degradations = {
        "jpeg_q30": lambda img: apply_jpeg_compression(img, 30),
        "jpeg_q50": lambda img: apply_jpeg_compression(img, 50),
        "jpeg_q70": lambda img: apply_jpeg_compression(img, 70),
        "blur_s1": lambda img: apply_gaussian_blur(img, 1.0),
        "blur_s2": lambda img: apply_gaussian_blur(img, 2.0),
        "down_up": lambda img: apply_down_up_sample(img, (112, 112))
    }
    
    for deg_name, deg_func in degradations.items():
        deg_out_dir = os.path.join(output_root, deg_name)
        print(f"[*] Đang tạo biến thể: {deg_name} tại {deg_out_dir}...")
        
        for root, _, files in os.walk(input_dir):
            for file in tqdm(files, desc=deg_name):
                if file.lower().endswith(('.jpg', '.jpeg', '.png')):
                    rel_path = os.path.relpath(root, input_dir)
                    target_dir = os.path.join(deg_out_dir, rel_path)
                    os.makedirs(target_dir, exist_ok=True)
                    
                    src_file = os.path.join(root, file)
                    dst_file = os.path.join(target_dir, os.path.splitext(file)[0] + ".jpg")
                    
                    try:
                        with Image.open(src_file) as img:
                            img = img.convert("RGB")
                            deg_img = deg_func(img)
                            deg_img.save(dst_file, "JPEG", quality=95)
                    except Exception as e:
                        print(f"[!] Lỗi tại file {src_file}: {e}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--input_dir", type=str, required=True, help="Thư mục tập test sạch")
    parser.add_argument("--output_root", type=str, required=True, help="Thư mục xuất ảnh suy biến")
    args = parser.parse_args()
    process_dataset(args.input_dir, args.output_root)
```

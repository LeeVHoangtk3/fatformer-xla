## 3. HƯỚNG DẪN THỰC HIỆN CHI TIẾT (STEP-BY-STEP)

### Cách triển khai giao diện Gradio (`demo/app.py`):
```python
import torch
from PIL import Image
import gradio as gr
from models.fatformer import build_fatformer_srm
from datasets.transforms import get_eval_transforms

# Khởi tạo mô hình
device = "cuda" if torch.cuda.is_available() else "cpu"
model = build_fatformer_srm(pretrained_clip="ViT-L-14.pt")
ckpt = torch.load("checkpoints/fatformer_srm_robust_final.pth", map_location=device)
model.load_state_dict(ckpt["model"], strict=True)
model.eval().to(device)

transforms = get_eval_transforms()

def predict(input_img):
    if input_img is None:
        return "Vui lòng chọn ảnh!", 0.0, 0.0
    
    img = Image.fromarray(input_img).convert("RGB")
    tensor = transforms(img).unsqueeze(0).to(device)
    
    with torch.no_grad():
        logits, lambda_val = model(tensor, return_lambda=True)
        prob = torch.softmax(logits, dim=-1)[0]
        fake_prob = prob[1].item() * 100
        real_prob = prob[0].item() * 100
        
    label = "🚨 FAKE (Ảnh Sinh Bởi AI)" if fake_prob > 50 else "✅ REAL (Ảnh Chụp Thật)"
    conf = fake_prob if fake_prob > 50 else real_prob
    
    status_text = f"Kết luận: {label}\nĐộ tin cậy: {conf:.2f}%\nHệ số cổng tần số λ(x): {lambda_val:.3f}"
    return status_text, {"Real": real_prob / 100, "Fake": fake_prob / 100}

demo = gr.Interface(
    fn=predict,
    inputs=gr.Image(label="Tải ảnh lên kiểm tra (Facebook/Telegram/Web)"),
    outputs=[gr.Textbox(label="Kết quả phân tích"), gr.Label(label="Xác suất chi tiết")],
    title="🛡️ FATFORMER FORENSIC DETECTOR (BTL Xử Lý Ảnh)",
    description="Hệ thống phát hiện ảnh AI bền vững trước nén mạng xã hội sử dụng SRM 3-Kernels và Dynamic Frequency Gating."
)

if __name__ == "__main__":
    demo.launch(share=True) # Tạo link public demo trực tiếp trong buổi thuyết trình
```

```python
import io
import random
from PIL import Image, ImageFilter
import torchvision.transforms as T
import torch

class RandomJPEGCompression:
    """Nén JPEG ngẫu nhiên với Quality trong khoảng [q_min, q_max]."""
    def __init__(self, q_min=40, q_max=95):
        self.q_min = q_min
        self.q_max = q_max

    def __call__(self, img: Image.Image) -> Image.Image:
        q = random.randint(self.q_min, self.q_max)
        buffer = io.BytesIO()
        img.save(buffer, format="JPEG", quality=q)
        buffer.seek(0)
        return Image.open(buffer).copy()

class RandomGaussianBlur:
    """Làm mờ ngẫu nhiên với sigma trong khoảng [s_min, s_max]."""
    def __init__(self, s_min=0.5, s_max=2.0, p=0.5):
        self.s_min = s_min
        self.s_max = s_max
        self.p = p

    def __call__(self, img: Image.Image) -> Image.Image:
        if random.random() < self.p:
            sigma = random.uniform(self.s_min, self.s_max)
            return img.filter(ImageFilter.GaussianBlur(radius=sigma))
        return img

class RandomDownUpSample:
    """Mô phỏng nén resize mạng xã hội."""
    def __init__(self, scale_range=(0.5, 0.75), p=0.3):
        self.scale_range = scale_range
        self.p = p

    def __call__(self, img: Image.Image) -> Image.Image:
        if random.random() < self.p:
            orig_size = img.size
            scale = random.uniform(*self.scale_range)
            inter_size = (int(orig_size[0] * scale), int(orig_size[1] * scale))
            down = img.resize(inter_size, Image.Resampling.BICUBIC)
            return down.resize(orig_size, Image.Resampling.BICUBIC)
        return img

class DualStreamRobustAugmentation:
    """
    Pipeline 2 Luồng:
    - 30% ảnh giữ Clean
    - 70% ảnh áp dụng suy biến vật lý ngẫu nhiên
    """
    def __init__(self, clean_prob=0.3):
        self.clean_prob = clean_prob
        self.jpeg = RandomJPEGCompression(q_min=40, q_max=95)
        self.blur = RandomGaussianBlur(s_min=0.5, s_max=2.0, p=0.5)
        self.down_up = RandomDownUpSample(scale_range=(0.5, 0.75), p=0.3)
        
        # Chuẩn hóa đầu vào CLIP ViT-L/14 (224x224, ImageNet mean/std)
        self.final_transform = T.Compose([
            T.Resize(256, interpolation=T.InterpolationMode.BICUBIC),
            T.CenterCrop(224),
            T.RandomHorizontalFlip(p=0.5),
            T.ToTensor(),
            T.Normalize(mean=[0.48145466, 0.4578275, 0.40821073], 
                        std=[0.26862954, 0.26130258, 0.27577711])
        ])

    def __call__(self, img: Image.Image) -> torch.Tensor:
        img = img.convert("RGB")
        # Luồng 2: Áp dụng suy biến với xác suất 70%
        if random.random() > self.clean_prob:
            img = self.jpeg(img)
            img = self.blur(img)
            img = self.down_up(img)
        # Áp dụng tiền xử lý chuẩn hóa hình học và tensor
        return self.final_transform(img)

def get_train_transforms():
    return DualStreamRobustAugmentation(clean_prob=0.3)

def get_eval_transforms():
    """Pipeline đánh giá chuẩn (chỉ crop tâm và chuẩn hóa)."""
    return T.Compose([
        T.Resize(256, interpolation=T.InterpolationMode.BICUBIC),
        T.CenterCrop(224),
        T.ToTensor(),
        T.Normalize(mean=[0.48145466, 0.4578275, 0.40821073], 
                    std=[0.26862954, 0.26130258, 0.27577711])
    ])
```
